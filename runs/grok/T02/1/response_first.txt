def labeled_crop(im, box, scale, step, path):
    x0,y0,x1,y1 = box
    crop = im.crop(box).resize(((x1-x0)*scale, (y1-y0)*scale), Image.NEAREST)
    margin = 36
    canvas = Image.new('RGB', (crop.width+margin, crop.height+margin), (255,255,255))
    canvas.paste(crop, (margin, margin))
    d = ImageDraw.Draw(canvas)
