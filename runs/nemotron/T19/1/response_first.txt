```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Veil Animation</title>
    <style>
        body {
            margin: 0;
            overflow: hidden;
            background-color: white;
        }
        canvas {
            display: block;
            width: 1200px;
            height: 600px;
        }
    </style>
</head>
<body>
    <canvas id="canvas" width="1200" height="600"></canvas>

    <script type="module">
        import * as THREE from './vendor/three.module.js';

        let renderer, scene, camera, mesh, texture;
        let material;
        let ready = false;

        const width = 1200;
        const height = 600;

        function init() {
            const canvas = document.getElementById('canvas');

            renderer = new THREE.WebGLRenderer({ canvas: canvas, antialias: false });
            renderer.setSize(width, height);
            renderer.setPixelRatio(1);
            renderer.toneMapping = THREE.NoToneMapping;

            scene = new THREE.Scene();

            camera = new THREE.OrthographicCamera(-width / 2, width / 2, height / 2, -height / 2, -1, 1);
            camera.position.z = 1;

            const loader = new THREE.TextureLoader();
            texture = loader.load('./image.png', onTextureLoaded);
            texture.colorSpace = THREE.SRGBColorSpace;

            material = new THREE.ShaderMaterial({
                uniforms: {
                    uMap: { value: texture },
                    uTime: { value: 0.0 }
                },
                vertexShader: `
                    varying vec2 vUv;
                    uniform float uTime;
                    void main() {
                        vUv = uv;
                        vec3 pos = position;

                        // Define veil region: Right side of the image.
                        // Image coords: x > 850 -> pos.x > 250 (since center is 0)
                        // We restrict vertical range to avoid shoes and focus on the drape.
                        float mask = 0.0;

                        if (pos.x > 220.0) {
                            // Exclude the face area roughly: image x: 900-1050, y: 100-300
                            // pos coords: x: 300-450, y: 200 down to 0.
                            bool isFace = (pos.x > 290.0 && pos.x < 460.0 && pos.y > -20.0 && pos.y < 220.0);
                            
                            // Exclude the main body/dress area below the chest to keep dress static
                            bool isBody = (pos.y < 50.0 && pos.x > 280.0);

                            if (!isFace && !isBody) {
                                mask = 1.0;
                            }
                        }

                        // Smooth falloff at the left edge of the veil zone to prevent shearing
                        float edgeFalloff = smoothstep(220.0, 240.0, pos.x);
                        mask *= edgeFalloff;

                        // Flutter animation: Horizontal wave based on vertical position and time
                        // Frequency varies slightly with Y to simulate fabric movement
                        float wave = sin(uTime * 3.14159265 + pos.y * 0.015) * 6.0;
                        
                        // Apply displacement only in X axis
                        pos.x += wave * mask;

                        gl_Position = projectionMatrix * modelViewMatrix * vec4(pos, 1.0);
                    }
                `,
                fragmentShader: `
                    varying vec2 vUv;
                    uniform sampler2D uMap;
                    void main() {
                        gl_FragColor = texture2D(uMap, vUv);
                    }
                `,
                transparent: false,
                depthTest: false,
                depthWrite: false
            });

            const geometry = new THREE.PlaneGeometry(width, height);
            mesh = new THREE.Mesh(geometry, material);
            scene.add(mesh);
        }

        function onTextureLoaded() {
            ready = true;
            window.ANIM_READY = true;
            // Render initial frame
            if (typeof window.setTime === 'function') {
                // setTime might be called before ready, but we ensure render here too.
            }
            render(0.0);
        }

        function render(t) {
            if (!texture) return;
            material.uniforms.uTime.value = t;
            renderer.render(scene, camera);
        }

        window.setTime = function(t) {
            // Clamp t to 0-2 just in case, though contract says 0 <= t <= 2
            const time = Math.max(0, Math.min(2, t));
            render(time);
        };

        init();

        // Optional: Live playback loop
        function animate() {
            if (ready) {
                // Loop time between 0 and 2
                const t = (performance.now() / 1000) % 2.0;
                render(t);
            }
            requestAnimationFrame(animate);
        }
        animate();

    </script>
</body>
</html>
```
